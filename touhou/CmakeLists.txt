#需求的最低cmake程序版本
cmake_minimum_required(VERSION 3.12)


#本工程的名字
project(OpenGL_Lecture)

#本工程支持的C++版本
set(CMAKE_CXX_STANDARD 17)

file(GLOB ASSETS "./assets" )

file(COPY ${ASSETS} DESTINATION ${CMAKE_BINARY_DIR})

include_directories(
	SYSTEM ${CMAKE_CURRENT_SOURCE_DIR}/thirdParty/include
)

link_directories(
	SYSTEM ${CMAKE_CURRENT_SOURCE_DIR}/thirdParty/lib
)
add_subdirectory(application)
#本工程所有cpp文件编译链接，生成exe
add_executable(touhou "main.cpp"  "glad.c" "main.h"  "application/texture/stb_image.h"  "game.cpp" "game.h" "math.cpp" "math.h" "application/media/threadManager.h" "application/media/MediaManager.h" "card.h" "card.cpp""girlish.cpp" "girlish.h" "application/button/button.h" "application/button/button.cpp" "application/player/player.h" "application/player/player.cpp" "application/enemy/enemy.h" "application/enemy/enemy.cpp" "application/bullet/bullet.cpp" "application/bullet/bullet.h")
target_link_libraries(touhou glfw3.lib shader uv vbo ebo vao texture media Uuid.lib)
